TRIGGER: UpdateBalanceAfterTransaction AFTER INSERT
Fires automatically after every INSERT into TRANSACTIONS. It sets ACCOUNT.Balance = NEW.BalanceAfter — no manual UPDATE needed. 
CREATE TRIGGER UpdateBalanceAfterTransaction
AFTER INSERT ON TRANSACTIONS
FOR EACH ROW
BEGIN
    UPDATE ACCOUNT
    SET    Balance = NEW.BalanceAfter
    WHERE  AccountID = NEW.AccountID;
END;

Stored Procedure: TransferMoney() Atomic Transaction
SQLite has no native stored procedures. TransferMoney() is implemented as a Python function using an atomic SQLite transaction. The MySQL-equivalent DDL is shown below.

MySQL Stored Procedure

DELIMITER $$
CREATE PROCEDURE TransferMoney(
    IN  p_from INT,  IN  p_to  INT,
    IN  p_amt  DECIMAL(15,2),
    OUT p_msg  VARCHAR(255)
)
BEGIN
  DECLARE v_bal DECIMAL(15,2);
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN ROLLBACK; SET p_msg='Error — rolled back'; END;

  START TRANSACTION;
  SELECT Balance INTO v_bal FROM ACCOUNT
  WHERE AccountID=p_from AND Status='Active' FOR UPDATE;

  IF v_bal IS NULL THEN
      ROLLBACK; SET p_msg='Not found/inactive';
  ELSEIF v_bal < p_amt THEN
      ROLLBACK; SET p_msg='Insufficient balance';
  ELSE
      -- Debit (trigger updates Balance automatically)
      INSERT INTO TRANSACTIONS(AccountID,TransactionType,
        Amount,Description,BalanceAfter)
      VALUES(p_from,'Transfer',p_amt,
        CONCAT('To AccID ',p_to), v_bal - p_amt);

      -- Credit (trigger updates Balance automatically)
      INSERT INTO TRANSACTIONS(AccountID,TransactionType,
        Amount,Description,BalanceAfter)
      VALUES(p_to,'Deposit',p_amt,
        CONCAT('From AccID ',p_from),
        (SELECT Balance FROM ACCOUNT
         WHERE AccountID=p_to) + p_amt);

      COMMIT; SET p_msg='Transfer successful';
  END IF;
END$$
DELIMITER ;

Python Equivalent (database.py)

def transfer_money(from_id, to_id, amount):
    conn = get_db()
    cur  = conn.cursor()
    try:
        src = cur.execute(
            "SELECT Balance, Status FROM ACCOUNT"
            " WHERE AccountID=?", (from_id,)).fetchone()

        # Validations
        if not src:
            return {"success":False,
                    "message":"Source not found"}
        if src["Status"] != "Active":
            return {"success":False,
                    "message":"Account not active"}
        if src["Balance"] < amount:
            return {"success":False,
                    "message":"Insufficient balance"}

        # Debit — trigger fires automatically
        cur.execute("""INSERT INTO TRANSACTIONS(AccountID,
          TransactionType,Amount,Description,BalanceAfter)
          VALUES(?,?,?,?,?)""",
          (from_id,'Transfer',amount,'...',
           src["Balance"]-amount))

        # Credit — trigger fires automatically

          TransactionType,Amount,Description,BalanceAfter)
